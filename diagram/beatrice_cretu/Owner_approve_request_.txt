sequenceDiagram
    actor Owner
    participant View as Others.cshtml
    participant GuiCtrl as RequestsController.cs (GUI-BRAP)
    participant Proxy as RequestProxyService.cs
    participant ApiCtrl as RequestsController.cs (API)
    participant ReqSvc as RequestService.cs
    participant ReqRepo as RequestRepository.cs
    participant NotifSvc as NotificationService.cs
    participant NotifRepo as NotificationRepository.cs
    participant DB as AppDbContext.cs

    Note over Owner,View: Offer = approve request and create rental

    Owner->>View: click Offer button
    View->>View: show.bs.modal<br/>set offerRequestId and offerForm.action
    Owner->>View: submit offerForm
    View->>GuiCtrl: Offer(int id)<br/>HTTP POST /Requests/Offer/{id}
    GuiCtrl->>Proxy: OfferGameAsync(id, RequestActionDataTransferObject)
    Proxy->>ApiCtrl: PUT /api/requests/{requestId}/offer
    ApiCtrl->>ReqSvc: OfferGame(requestId, body.AccountId)

    ReqSvc->>ReqRepo: Get(requestId)
    ReqRepo->>DB: RequestsWithNavigations(...).FirstOrDefault(...)
    DB-->>ReqRepo: request / null
    ReqRepo-->>ReqSvc: req / KeyNotFoundException

    alt request not found / wrong owner / request not open
        ReqSvc-->>ApiCtrl: Failure(...)
        ApiCtrl-->>Proxy: 4xx error
        Proxy-->>GuiCtrl: ProxyServiceException
        GuiCtrl-->>View: RedirectToAction(nameof(Others))
    else valid open request
        Note over ReqSvc: service computes buffered dates
        ReqSvc->>ReqRepo: GetOverlappingRequests(gameId, req.Id, buffStart, buffEnd)
        ReqRepo->>DB: RequestsWithNavigations(...).Where(...).ToImmutableList()
        DB-->>ReqRepo: conflicts
        ReqRepo-->>ReqSvc: conflicts

        alt atomic approval succeeds
            ReqSvc->>ReqRepo: ApproveAtomically(request, conflicts)
            Note over ReqRepo,DB: one transaction updates the data
            ReqRepo->>DB: BeginTransaction()<br/>ExecuteDelete()<br/>Rentals.Add(newRental)<br/>SaveChanges()<br/>Commit()
            DB-->>ReqRepo: rentalId
            ReqRepo-->>ReqSvc: rentalId

            loop for each conflicting request
                ReqSvc->>NotifSvc: SendNotificationToUser(conflictRenterId, NotificationDTO)
                NotifSvc->>NotifRepo: Add(notification)
                NotifRepo->>DB: Notifications.Add(notification)<br/>SaveChanges()
                DB-->>NotifRepo: saved
            end

            ReqSvc->>NotifSvc: SendNotificationToUser(renterId, NotificationDTO)
            NotifSvc->>NotifRepo: Add(notification)
            NotifRepo->>DB: Notifications.Add(notification)<br/>SaveChanges()
            DB-->>NotifRepo: saved

            ReqSvc-->>ApiCtrl: Success(rentalId)
            ApiCtrl-->>Proxy: 200 OK { RentalId }
            Proxy-->>GuiCtrl: return
            GuiCtrl-->>View: RedirectToAction(nameof(Others))

            View->>GuiCtrl: Others()<br/>HTTP GET /Requests/Others
            GuiCtrl->>Proxy: GetOpenRequestsForOwnerAsync(ownerAccountId)
            Proxy->>ApiCtrl: GET /api/requests/owner/{ownerAccountId}/open
            ApiCtrl->>ReqSvc: GetOpenRequestsForOwner(ownerAccountId)
            ReqSvc->>ReqRepo: GetRequestsByOwner(ownerAccountId)
            ReqRepo->>DB: RequestsWithNavigations(...).Where(...).ToImmutableList()
            DB-->>ReqRepo: requests
            ReqRepo-->>ReqSvc: requests
            ReqSvc-->>ApiCtrl: openRequests
            ApiCtrl-->>Proxy: 200 OK [RequestDTO]
            Proxy-->>GuiCtrl: openRequests
            GuiCtrl-->>View: View(openRequests)
        else atomic approval fails
            ReqSvc-->>ApiCtrl: Failure(OfferError.TransactionFailed)
            ApiCtrl-->>Proxy: 409 Conflict
            Proxy-->>GuiCtrl: ProxyServiceException
            GuiCtrl-->>View: RedirectToAction(nameof(Others))
        end
    end
