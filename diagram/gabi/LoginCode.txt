@startuml
  actor User

  participant "AuthController (GUI-BRAP)" as WebCtrl
  participant "AuthProxyServiceAdapter (GUI-BRAP)" as ProxyAdapter
  participant "AuthService (ApiClient)" as ApiClientSvc
  participant "AuthController (Api)" as ApiCtrl
  participant "AuthService (Api)" as ApiSvc
  participant "AccountRepository (Api)" as AccountRepo
  participant "FailedLoginRepository (Api)" as FailedRepo
  database "SQL Server" as DB

  User -> WebCtrl : POST /login (LoginViewModel)
  WebCtrl -> ProxyAdapter : LoginAsync(LoginDataTransferObject)
  ProxyAdapter -> ApiClientSvc : LoginAsync(LoginDataTransferObject)
  ApiClientSvc -> ApiCtrl : POST api/auth/login (HTTP)
  ApiCtrl -> ApiSvc : LoginAsync(LoginDataTransferObject)

  ApiSvc -> AccountRepo : GetByUsernameAsync(username) or GetByEmailAsync(email)
  AccountRepo -> DB : SELECT Accounts WHERE Username or Email = ?
  DB --> AccountRepo : Account
  AccountRepo --> ApiSvc : Account

  ApiSvc -> FailedRepo : GetByAccountIdAsync(accountId)
  FailedRepo -> DB : SELECT FailedLoginAttempts WHERE AccountId = ?
  DB --> FailedRepo : FailedLoginAttempt
  FailedRepo --> ApiSvc : FailedLoginAttempt


  ApiSvc -> FailedRepo : ResetAsync(accountId)
  FailedRepo -> DB : UPDATE FailedLoginAttempts SET FailedAttempts=0
  ApiSvc --> ApiCtrl : ServiceResult(AccountProfileDTO) success
  ApiCtrl --> ApiClientSvc : 200 OK + AccountProfileDTO
  ApiClientSvc --> ProxyAdapter : ServiceResult(AccountProfileDTO)
  ProxyAdapter --> WebCtrl : AccountProfileDTO
  WebCtrl -> WebCtrl : SignInAsync(ClaimsPrincipal)
  WebCtrl --> User : Redirect to /games

  @enduml
